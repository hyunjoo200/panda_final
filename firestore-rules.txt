rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ▼▼▼ 아래 따옴표 안을 선생님 로그인 이메일로 바꾸세요 ▼▼▼
    function isTeacher() {
      return request.auth != null && request.auth.token.email == 'teacher@example.com';
    }
    // ▲▲▲ 여기까지만 고치면 됩니다 ▲▲▲

    function okName(d)  { return d.name is string && d.name.size() >= 1 && d.name.size() <= 12; }
    function okScore(d) { return d.total is int && d.total >= 0 && d.total <= 5000; }

    // 학생 한 명의 자세한 결과: 본인과 선생님만 읽을 수 있어요
    match /rooms/{room}/players/{uid} {
      allow read: if isTeacher() || (request.auth != null && request.auth.uid == uid);
      allow create, update: if request.auth != null && request.auth.uid == uid
                            && okName(request.resource.data) && okScore(request.resource.data);
      allow delete: if isTeacher();
    }

    // 우리 반 취향 집계 (10차시~): 이름 없이 보기마다 "몇 명"만 들어 있어요.
    // 한 번에 한 보기만, 1씩만 올릴 수 있어요.
    function oneUp(newD, oldD) {
      let ks = newD.diff(oldD).affectedKeys();
      return ks.size() == 1 && newD[ks.toList()[0]] == oldD.get(ks.toList()[0], 0) + 1;
    }
    match /rooms/{room}/tally/{round} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.keys().size() == 1
                    && request.resource.data[request.resource.data.keys()[0]] == 1;
      allow update: if request.auth != null && request.resource.data.keys().size() <= 8
                    && oneUp(request.resource.data, resource.data);
      allow delete: if isTeacher();
    }

    // 순위표: 이름·번호·점수만 들어 있고, 게임에 들어온 학생이면 읽을 수 있어요
    match /rooms/{room}/board/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid
                            && okName(request.resource.data) && okScore(request.resource.data);
      allow delete: if isTeacher();
    }
  }
}
